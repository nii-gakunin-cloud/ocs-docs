class VcpSpecResourceGcp(VcpSpecResource):

    """
    サンプルコード

.. code-block:: python

    spec = sdk.get_spec("gcp", "small")
    #
    # 変更できること
    #
    # spec.num_nodes = 1
    # spec.params_v = ['/opt:/opt']
    # spec.params_e = ['USER_NAME=test']
    # spec.ip_addresses = ['起動するnodeの静的なIPアドレス']

    # Google Cloud Platform 依存
    #   https://www.terraform.io/docs/providers/google/index.html
    # spec.machine_type = 'n1-standard-1'  # f1-micro|n1-standard-1|...
    # spec.disk_size_gb = 40
    # spec.disk_type = "pd-standard"　# pd-standard|pd-sd|...

    # 追加で使用するVolume
    # spec.disks = ['vol-08cbb04b35c8c9545']

    # cloud上のGCPイメージ設定
    # spec.cloud_image = 'niivcp-20170616'

    # base containerにssh loginするためのssh公開鍵情報を設定
    spec.set_ssh_pubkey('tmp/id_rsa.pub')
    """

    version = CCI_VERSION + "+" + BUILD_VERSION

    def __init__(self, provider_name, flavor, config_dir):

        # flavor から spec を設定
        super().__init__(provider_name, flavor, config_dir)
        # spec attributes名の設定チェックの回避
        super().__setattr__("gcp_cci_", None)

        # credentail情報は、vcp_config から設定
        self.gcp_cci_ = {"cci_version": CCI_VERSION}
        self.gcp_cci_["credentials"] = self.vcp_config_["gcp"]["credentials"]
        self.unit_cci_["private_network"] = self.vcp_config_["gcp"][
            "private_network"
        ]  # VPNネットワーク名

        # flavorから初期化
        self.gcp_cci_["machine_type"] = self.flavor_["machine_type"]
        self.gcp_cci_["disk_size_gb"] = self.flavor_["disk_size_gb"]
        self.gcp_cci_["disk_type"] = self.flavor_["disk_type"]

        self.gcp_cci_["cloud_image"] = "default"

        # 初期化
        self.gcp_cci_["ip_address_list"] = []
        self.gcp_cci_["attached_disk"] = []

    def __str__(self):
        text = """
========================
{provider_name}
------------------------
{unit_info}""".format(
            provider_name=self.unit_cci_["cloud_provider"], unit_info=super().__str__()
        )

        text += """
    machine_type: {machine_type}
    disk_size_gb: {disk_size_gb}
    disk_type: {disk_type}
    attached_disk: {attached_disk}
    cloud_image: {cloud_image}
========================""".format_map(
            self.gcp_cci_
        )

        return text

    def cci(self, name):
        """
        CCI生成

        :param name: unit名

        :return: CCI文字列
        """
        # cloud parameter 情報
        my_cci = super().cci(name)

        # yaml schema check
        self.cci_schema_.validate(self.gcp_cci_, "gcp")

        # cloud_params
        my_cci += """
    cloud_provider: gcp
    cloud_params:
        cci_version: "{cci_version}"
        machine_type: {machine_type}
        disk_size_gb: {disk_size_gb}
        disk_type: {disk_type}
        attached_disk: {attached_disk}
        cloud_image: {cloud_image}
        ip_address_list: {ip_address_list}
        credentials: {credentials}""".format_map(
            self.gcp_cci_
        )

        return my_cci

    # property: ip_addresses

    def ip_addresses(self):
        """
        GCPのVPC上の静的ip_address_list
        """
        return self.gcp_cci_["ip_address_list"]

    @ip_addresses.setter
    def ip_addresses(self, v):
        """
        GCPのVPC上の静的ip_address_list

        .. note::
            `num_nodes` と `ip_addresses` は最後にセットされたほうが有効になる。
        """
        if 0 < len(v):
            self.unit_cci_["num_nodes"] = len(v)
        self.gcp_cci_["ip_address_list"] = v

    # property: cloud_image
    @property
    def cloud_image(self):
        """
        GCPに依存するcloud_image
        """
        return self.gcp_cci_["cloud_image"]

    @cloud_image.setter
    def cloud_image(self, v):
        self.gcp_cci_["cloud_image"] = v

    @VcpSpecResource.num_nodes.setter
    def num_nodes(self, v):
        """
        起動するnodeの個数

        .. note::
            `num_nodes` と `ip_addresses` は最後にセットされたほうが有効になる。
        """
        self.unit_cci_["num_nodes"] = v
        self.gcp_cci_["ip_address_list"] = []

    def __setattr__(self, name, value):
        """
        spec の attributes に不正な値を設定した場合に例外を発生
        :param name: attributes名
        :param value: 設定値
        """
        if not hasattr(self, name):
            raise AttributeError("spec resource has no attributes!!  [{}]".format(name))
        super().__setattr__(name, value)

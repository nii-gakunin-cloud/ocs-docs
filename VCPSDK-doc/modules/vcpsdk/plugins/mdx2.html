class VcpSpecResourceMdx2(VcpSpecResource):

    """
    サンプルコード

.. code-block:: python

    spec = sdk.get_spec("mdx2", "default")
    #
    # 変更できること
    #
    # spec.num_nodes = 1
    # spec.params_v = ['/opt/test:/opt/test']
    # spec.params_e = ['USER_NAME=test']
    # spec.ip_addresses = ['起動するnodeの静的なIPアドレス']

    # 追加で使用するVolume
    # spec.disks = ['vol-08cbb04b35c8c9545']    # volume_id 指定
    # or
    # my_disks = disk_unit.find_nodes()
    # spec.disks = [my_disks[0]]   # VC disk指定

    # プロバイダ依存パラメータ
    #   https://search.opentofu.org/provider/terraform-provider-openstack/openstack/latest
    # spec.flavor_name = 'vc1m2g'
    # spec.volume_size = 40
    # spec.volume_type = "project-volume" # project-volume|tripleo

    # cloud上のタグ設定
    # spec.set_tag('key1', 'value1')
    # spec.set_tag('key2', 'value2')

    # cloud上のイメージ設定
    # spec.cloud_image = '5fed0361-58f9-41d7-8281-64c9cd8efeb2'  # mdxII-Ubuntu-22.04-Server

    # base containerにssh loginするためのssh公開鍵情報を設定
    spec.set_ssh_pubkey('tmp/id_rsa.pub')
    """

    version = CCI_VERSION + "+" + BUILD_VERSION

    def __init__(self, provider_name, flavor, config_dir):

        # flavor から spec を設定
        super().__init__(provider_name, flavor, config_dir)

        # spec attributes名の設定チェックの回避
        super().__setattr__("mdx2_cci_", None)

        # credentail情報は、vcp_config から設定
        self.mdx2_cci_ = {"cci_version": CCI_VERSION}
        self.mdx2_cci_["user_name"] = self.vcp_config_[provider_name]["user_name"]
        self.mdx2_cci_["password"] = self.vcp_config_[provider_name]["password"]
        # VPNネットワーク名
        self.unit_cci_["private_network"] = self.vcp_config_[provider_name][
            "private_network"
        ]

        # flavorから初期化
        self.mdx2_cci_["flavor_name"] = self.flavor_["flavor_name"]
        self.mdx2_cci_["volume_size"] = self.flavor_["volume_size"]
        self.mdx2_cci_["volume_type"] = self.flavor_["volume_type"]

        self.mdx2_cci_["cloud_image"] = "default"

        # 初期化
        self.mdx2_cci_["ip_address_list"] = []
        self.mdx2_cci_["volume_id"] = []

    def __str__(self):
        text = """
========================
{provider_name}
------------------------
{unit_info}
""".format(
            provider_name=self.unit_cci_["cloud_provider"], unit_info=super().__str__()
        )

        text += """
    flavor_name: {flavor_name}
    volume_size: {volume_size}
    volume_type: {volume_type}
    volume_id: {volume_id}
    user_name: {user_name}
    cloud_image: {cloud_image}
    ip_address_list: {ip_address_list}
========================""".format_map(
            self.mdx2_cci_
        )
        text += """
    tags: {}
========================""".format(
            self.get_tag()
        )

        return text

    def cci(self, name):
        """
        CCI生成

        :param name: unit名

        :return: CCI文字列
        """
        # yaml schema check
        self.cci_schema_.validate(self.mdx2_cci_, self.unit_cci_["cloud_provider"])

        my_cci = super().cci(name)

        my_cci += """
    cloud_provider: {provider_name}
    cloud_params:""".format(
            provider_name=self.unit_cci_["cloud_provider"]
        )

        my_cci += """
        cci_version: "{cci_version}"
        flavor_name: {flavor_name}
        volume_size: {volume_size}
        volume_type: {volume_type}
        user_name: "{user_name}"
        password: {password}
        ip_address_list: {ip_address_list}
        volume_id: {volume_id}
        cloud_image: {cloud_image}""".format_map(
            self.mdx2_cci_
        )
        my_cci += self.cci_tags()

        return my_cci

    # property: disks
    @property
    def disks(self):
        """
        """
        return self.mdx2_cci_["volume_id"]

    @disks.setter
    def disks(self, v):
        # volume_id は、UUID形式 と VcNode の2種類がある
        volumes = []
        for volume in v:
            if isinstance(volume, vcplib.occtr.VcNode):
                # VcNode 指定
                volume_id = "vcp:{vc_id}:{unit_name}:{disk_id}".format(
                    vc_id=volume.vc.vcid, unit_name=volume.unit.name, disk_id=volume.id
                )
            else:
                volume_id = volume
            volumes.append(volume_id)

        self.mdx2_cci_["volume_id"] = volumes

    # property: type
    @property
    def volume_type(self):
        """
        MDX2に依存するvolume_type

        .. note::
            * VCP SDK flavorで設定可能
        """
        return self.mdx2_cci_["volume_type"]

    @volume_type.setter
    def volume_type(self, v):
        self.mdx2_cci_["volume_type"] = v

    # property: ip_addresses
    @property
    def ip_addresses(self):
        return self.mdx2_cci_["ip_address_list"]

    @ip_addresses.setter
    def ip_addresses(self, v):
        """
        静的ip_address_list

        .. note::
            `num_nodes` と `ip_addresses` は最後にセットされたほうが有効になる。
        """
        # num_nodesを上書きする
        if 0 < len(v):
            self.unit_cci_["num_nodes"] = len(v)
        self.mdx2_cci_["ip_address_list"] = v

        起動するnodeの個数

        .. note::
            `num_nodes` と `ip_addresses` は最後にセットされたほうが有効になる。
        """
        self.unit_cci_["num_nodes"] = v
        self.mdx2_cci_["ip_address_list"] = []
